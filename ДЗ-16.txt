using System;

class Program
{
    static void Main()
    {
        const int MinN = 10;
        const int MaxN = 25;
        const int StartNumber = 50;
        const int EndNumber = 150;

        Random random = new Random();
        int N = random.Next(MinN, MaxN + 1);

        Console.WriteLine($"N = {N}");

        int count = 0;
        int currentNumber = StartNumber;

       
        while (currentNumber <= EndNumber)
        {
            int remainder = currentNumber;

            
            while (remainder >= N)
            {
                remainder -= N;
            }

           
            if (remainder == 0)
            {
                count++;
                currentNumber += N; 
            }
            else
            {
                currentNumber++; 
            }
        }

        Console.WriteLine($"Количество чисел от {StartNumber} до {EndNumber} кратных {N}: {count}");
    }
}
