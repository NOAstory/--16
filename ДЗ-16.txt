using System;

class Program
{
    static void Main()
    {
        const int MinN = 10;
        const int MaxN = 25;
        const int StartNumber = 50;
        const int EndNumber = 150;

        Random random = new Random();
        int N = random.Next(MinN, MaxN + 1);

        Console.WriteLine($"N = {N}");

        int count = 0;

        
        int firstMultiple = StartNumber;
        while (firstMultiple % N != 0)  
        {
            firstMultiple++;
        }

        
        for (int number = firstMultiple; number <= EndNumber; number += N)
        {
            count++;
        }

        Console.WriteLine($"Количество чисел от {StartNumber} до {EndNumber} кратных {N}: {count}");
    }
}
