using System;

class Program
{
    static void Main()
    {
        const int MinValue = 10;
        const int MaxValue = 25;
        const int StartRange = 50;
        const int EndRange = 150;

        Random random = new Random();
        int divisor = random.Next(MinValue, MaxValue + 1);

        Console.WriteLine($"Случайное число: {divisor}");

        int count = 0;

        for (int number = StartRange; number <= EndRange; number++)
        {
            int temp = number;
            bool isMultiple = false;

            while (temp >= divisor)
            {
                temp -= divisor;
            }

            if (temp == 0)
            {
                isMultiple = true;
            }

            if (isMultiple)
            {
                count++;
            }
        }

        Console.WriteLine($"Количество чисел от {StartRange} до {EndRange} кратных {divisor}: {count}");
    }
}
