using System;

class Program
{
    static void Main()
    {
        
        Random random = new Random();
        int N = random.Next(10, 26); 

        Console.WriteLine($"N = {N}");

        int count = 0; 

        
        for (int i = 50; i <= 150; i++)
        {
           
            int current = i;
            while (current >= N)
            {
                current -= N;
            }

            
            if (current == 0)
            {
                count++;
            }
        }

        Console.WriteLine($"Количество чисел от 50 до 150 кратных {N}: {count}");
    }
}
